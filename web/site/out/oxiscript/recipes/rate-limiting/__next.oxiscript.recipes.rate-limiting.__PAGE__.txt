<h2>Rate limiting — atomic counters with TTL</h2>
<p>Per-key counters that auto-expire. Pair OxiScript with a TTL index on the bucket key for free cleanup.</p>

<h3>Schema</h3>
<pre><code class="lang-bash">rate_buckets: {key, count, window_start, ttl_at}
              # TTL index on ttl_at via create_ttl_index</code></pre>

<h3>check_rate</h3>
<pre><code class="lang-rust">proc check_rate(key, max_per_window, window_seconds) {
    let bucket = find_one(<span class="str">"rate_buckets"</span>, {key: key})
    <span class="kw">if</span> bucket == <span class="kw">null</span> {
        insert(<span class="str">"rate_buckets"</span>, {
            key: key, count: <span class="num">1</span>,
            window_start: <span class="str">"now"</span>,
            ttl_at: <span class="str">"now+window_seconds"</span>
        })
        <span class="kw">return</span> {allowed: <span class="kw">true</span>, count: <span class="num">1</span>, limit: max_per_window}
    }
    <span class="kw">if</span> bucket.count &gt;= max_per_window {
        <span class="kw">return</span> {allowed: <span class="kw">false</span>, count: bucket.count, limit: max_per_window}
    }
    update(<span class="str">"rate_buckets"</span>, {key: key}, {$inc: {count: <span class="num">1</span>}})
    <span class="kw">return</span> {allowed: <span class="kw">true</span>, count: bucket.count + <span class="num">1</span>, limit: max_per_window}
}</code></pre>
