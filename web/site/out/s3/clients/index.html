S3 · Use it From

Clients

Anything that talks AWS S3 talks to OxiDB. Pass --endpoint-url (or set endpoint_url in the SDK config), and you're done.

aws-cli

aws configure set aws_access_key_id minioadmin
aws configure set aws_secret_access_key minioadmin
aws configure set default.region us-east-1

ENDPOINT=http://localhost:9000
aws --endpoint-url $ENDPOINT s3 ls
aws --endpoint-url $ENDPOINT s3 mb s3://photos
aws --endpoint-url $ENDPOINT s3 cp ./local s3://photos/ --recursive
aws --endpoint-url $ENDPOINT s3 sync ./local s3://photos/

# Persist endpoint to skip the flag (~/.aws/config)
[profile oxidb]
endpoint_url = http://localhost:9000
region = us-east-1

aws --profile oxidb s3 ls

Python — boto3

import boto3

s3 = boto3.client(
    "s3",
    endpoint_url="http://localhost:9000",
    aws_access_key_id="minioadmin",
    aws_secret_access_key="minioadmin",
    region_name="us-east-1"
)

s3.create_bucket(Bucket="photos")
s3.upload_file("avatar.png", "photos", "me.png")
s3.download_file("photos", "me.png", "local.png")
for page in s3.get_paginator("list_objects_v2").paginate(Bucket="photos"):
    for o in page.get("Contents", []):
        print(o["Key"], o["Size"])

Python — s3fs

import s3fs

fs = s3fs.S3FileSystem(
    key="minioadmin", secret="minioadmin",
    client_kwargs={"endpoint_url": "http://localhost:9000"}
)

with fs.open("photos/me.png", "wb") as f:
    f.write(image_bytes)

for path in fs.ls("photos"):
    print(path)

mc — MinIO client

mc alias set oxidb http://localhost:9000 minioadmin minioadmin

mc mb oxidb/photos

mc ls oxidb/photos/
mc cat oxidb/photos/me.png > recovered.png
mc rm oxidb/photos/me.png
mc rb oxidb/photos
mc mirror ./local oxidb/photos/

JavaScript — AWS SDK v3

import { S3Client, PutObjectCommand, GetObjectCommand } from "@aws-sdk/client-s3"

const s3 = new S3Client({
  endpoint: "http://localhost:9000",
  region: "us-east-1",
  credentials: { accessKeyId: "minioadmin", secretAccessKey: "minioadmin" },
  forcePathStyle: true     // IMPORTANT — OxiDB uses path-style URLs
})

await s3.send(new PutObjectCommand({
  Bucket: "photos", Key: "me.png", Body: imageBuffer,
  ContentType: "image/png"
}))

const obj = await s3.send(new GetObjectCommand({
  Bucket: "photos", Key: "me.png"
}))

Go — aws-sdk-go-v2

import (
    "context"
    "github.com/aws/aws-sdk-go-v2/aws"
    "github.com/aws/aws-sdk-go-v2/credentials"
    "github.com/aws/aws-sdk-go-v2/service/s3"
)

cfg := aws.Config{
    Region: "us-east-1",
    Credentials: credentials.NewStaticCredentialsProvider("minioadmin", "minioadmin", ""),
    BaseEndpoint: aws.String("http://localhost:9000"),
}

client := s3.NewFromConfig(cfg, func(o *s3.Options) {
    o.UsePathStyle = true
})

client.PutObject(context.TODO(), &s3.PutObjectInput{
    Bucket: aws.String("photos"),
    Key:    aws.String("me.png"),
    Body:   bytes.NewReader(data),
})

rclone

# ~/.config/rclone/rclone.conf
[oxidb]
type = s3
provider = Other
access_key_id = minioadmin
secret_access_key = minioadmin
endpoint = http://localhost:9000
force_path_style = true

rclone copy ./local oxidb:photos/
rclone sync oxidb:photos/ ./backup/
rclone ls oxidb:photos

Browser uploads — presigned URL + fetch

// Backend (boto3)
url = s3.generate_presigned_url("put_object",
  Params={"Bucket": "uploads", "Key": "u-1.png", "ContentType": "image/png"},
  ExpiresIn=600)

// Frontend (any modern browser)
await fetch(url, {
  method: "PUT",
  body: file,
  headers: { "Content-Type": "image/png" }
})
Path style vs virtual host: OxiDB only speaks path-style URLs (http://endpoint/bucket/key). Force this in any SDK that defaults to virtual-host style — the option is usually called force_path_style or UsePathStyle.
